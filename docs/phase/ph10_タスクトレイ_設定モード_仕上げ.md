#### 2-2. 仮想設定ページの生成

設定タイルを定義して `PageViewModel` を動的生成する。`AppConfig` の `TileCountCols` から 1 行あたりの最大タイル数を求め、複数ページになる場合はページを分割する。

追加フィールド：

```csharp
private int _settingsPageIndex;
public int SettingsPageIndex
{
    get => _settingsPageIndex;
    set { _settingsPageIndex = Math.Clamp(value, 0, SettingsPages.Count - 1); OnPropertyChanged(); OnPropertyChanged(nameof(DisplayPage)); }
}

// 設定タイル定義 ─ 追加のたびここにエントリを足す
private static readonly (string title, string path)[] SettingsTileDefs =
[
    ("全体設定", "settings://globalSettings"),
];
```

設定ページ群を生成するヘルパー：

```csharp
public List<PageViewModel> SettingsPages { get; } = [];

private void BuildSettingsPages()
{
    SettingsPages.Clear();
    int cols = App.ConfigService.Current.Global.TileCountCols;
    int rows = App.ConfigService.Current.Global.TileCountRows;
    int tilesPerPage = cols * rows;

    for (int start = 0; start < SettingsTileDefs.Length; start += tilesPerPage)
    {
        var pageConfig = new PageConfig { Name = "詳細設定" };
        for (int i = start; i < Math.Min(start + tilesPerPage, SettingsTileDefs.Length); i++)
        {
            var (title, path) = SettingsTileDefs[i];
            int localIdx = i - start;
            pageConfig.Tiles.Add(new TileConfig
            {
                Col      = localIdx % cols,
                Row      = localIdx / cols,
                ColSpan  = 1,
                RowSpan  = 1,
                Type     = "settings",  // 専用タイプ
                Title    = title,
                Path     = path,
                Color    = "gray",
                Opacity  = 0,
                FontSizePt = 16,
                FontColor  = "white",
            });
        }
        SettingsPages.Add(new PageViewModel(pageConfig));
    }
}
```

コンストラクタの末尾で `BuildSettingsPages()` を呼ぶ。

**タイル配置数の縮小制限**（`ApplyToConfig` 内で適用前にチェック）：

```csharp
// 縮小制限チェック：配置済みタイルの最大座標を下回れない
int minCols = App.ConfigService.Current.Pages
    .SelectMany(p => p.Tiles)
    .Select(t => t.Col + t.ColSpan)
    .DefaultIfEmpty(1).Max();
int minRows = App.ConfigService.Current.Pages
    .SelectMany(p => p.Tiles)
    .Select(t => t.Row + t.RowSpan)
    .DefaultIfEmpty(1).Max();
g.TileCountCols = Math.Max(TileCountCols, minCols);
g.TileCountRows = Math.Max(TileCountRows, minRows);
```

#### 9-2. 4方向吸着対応

`ApplySnap()` を方向ごとに分岐させる。`_direction` フィールドを追加して現在の吸着方向を保持する。

```csharp
private string _direction = "right"; // "right" | "left" | "top" | "bottom"

public void ApplySnap()
{
    var config = App.ConfigService.Current;
    var layout = config.Layout;
    var global = config.Global;
    _direction = global.SnapPosition;

    var (frameW, frameH) = WindowSizeCalculator.Calculate(global, layout);

    // モニター情報取得（マルチモニター § 9-3 参照）
    var monitor = GetTargetMonitor(global.SnapMonitor);
    double screenLeft = monitor.Left;
    double screenTop  = monitor.Top;
    double screenW    = monitor.Width;
    double screenH    = monitor.Height;

    switch (_direction)
    {
        case "right":
            _window.Width  = layout.HandleShortSide + layout.HandleFrameMargin + frameW;
            _baseFrameH    = frameH;
            _normalLeft    = screenLeft + screenW - layout.ScreenEdgeDistance - frameW
                             - layout.HandleFrameMargin - layout.HandleShortSide;
            _storedLeft    = _normalLeft + frameW + layout.HandleFrameMargin;
            _window.Left   = _normalLeft;
            _window.Top    = screenTop + (screenH - _window.Height) / 2;
            break;

        case "left":
            _window.Width  = layout.HandleShortSide + layout.HandleFrameMargin + frameW;
            _baseFrameH    = frameH;
            // 左吸着：フレームが左側、ハンドルが右側（XAML の列順を逆にする必要あり）
            _normalLeft    = screenLeft + layout.ScreenEdgeDistance;
            _storedLeft    = _normalLeft - frameW - layout.HandleFrameMargin;
            _window.Left   = _normalLeft;
            _window.Top    = screenTop + (screenH - _window.Height) / 2;
            break;

        case "top":
            // TODO: 上吸着（Ph.10 以降で実装）
            break;

        case "bottom":
            // TODO: 下吸着（Ph.10 以降で実装）
            break;
    }

    UpdateWindowHeight();
}
```

> **実装注意（左吸着）**：右吸着の XAML では `列0 = ハンドル`、`列2 = フレーム` の順。左吸着では `列0 = フレーム`、`列2 = ハンドル` が自然だが、XAML を動的に切り替えるのは複雑。初期フェーズは「右吸着」「左吸着」の 2 方向のみ実装し、上下は TODO としても構わない。

#### 9-3. マルチモニター対応

Win32 `EnumDisplayMonitors` を使ってモニター情報を取得する。

```csharp
[DllImport("user32.dll")]
private static extern bool EnumDisplayMonitors(IntPtr hdc, IntPtr lprcClip,
    MonitorEnumProc lpfnEnum, IntPtr dwData);
private delegate bool MonitorEnumProc(IntPtr hMonitor, IntPtr hdcMonitor,
    ref RECT lprcMonitor, IntPtr dwData);

[DllImport("user32.dll", CharSet = CharSet.Auto)]
private static extern bool GetMonitorInfo(IntPtr hMonitor, ref MONITORINFO lpmi);

[StructLayout(LayoutKind.Sequential)]
private struct RECT { public int Left, Top, Right, Bottom; }

[StructLayout(LayoutKind.Sequential, CharSet = CharSet.Auto)]
private struct MONITORINFO
{
    public int    cbSize;
    public RECT   rcMonitor;
    public RECT   rcWork;
    public uint   dwFlags;
}

private record MonitorRect(double Left, double Top, double Width, double Height);

private MonitorRect GetTargetMonitor(int snapMonitor)
{
    var monitors = new List<RECT>();
    EnumDisplayMonitors(IntPtr.Zero, IntPtr.Zero, (hMon, _, ref rc, _) =>
    {
        monitors.Add(rc);
        return true;
    }, IntPtr.Zero);

    // snapMonitor は 1 始まり
    int idx = Math.Clamp(snapMonitor - 1, 0, monitors.Count - 1);
    var rc  = monitors[idx];

    // 物理ピクセル → 論理ピクセル変換
    double dpiX = GetDpiScaleX();
    double dpiY = GetDpiScaleY();
    return new MonitorRect(
        rc.Left   / dpiX,
        rc.Top    / dpiY,
        (rc.Right  - rc.Left) / dpiX,
        (rc.Bottom - rc.Top)  / dpiY);
}

private double GetDpiScaleX()
{
    var source = PresentationSource.FromVisual(_window);
    return source?.CompositionTarget.TransformToDevice.M11 ?? 1.0;
}
```

#### 9-4. ドラッグ方向の方向別切替

現在は `OnMouseMove` で Y 方向のみ動く。吸着方向が `"top"` / `"bottom"` のときは X 方向のみ動くよう分岐する。

```csharp
// OnMouseMove 内、ドラッグ中の計算部分
double dpiScaleX = GetDpiScaleX();
double dpiScaleY = GetDpiScaleY();
double currentX = _window.PointToScreen(e.GetPosition(_window)).X;
double currentY = _window.PointToScreen(e.GetPosition(_window)).Y;

if (_direction is "right" or "left")
{
    // 縦スライド
    double newTop  = _dragStartTop + (currentY - _dragStartMouseY) / dpiScaleY;
    _window.Top    = Math.Clamp(newTop, 0, SystemParameters.PrimaryScreenHeight - _window.Height);
}
else
{
    // 横スライド（top / bottom 吸着）
    double newLeft = _dragStartLeft + (currentX - _dragStartMouseX) / dpiScaleX;
    _window.Left   = Math.Clamp(newLeft, 0, SystemParameters.PrimaryScreenWidth - _window.Width);
}
```

`_dragStartLeft` / `_dragStartMouseX` フィールドを追加し、`OnMouseDown` でも X 座標を記録する。

### 10. TrayIconService（新規）

```csharp
using System.Windows;
using Hardcodet.Wpf.TaskbarNotification;
using AppLauncher.Models;
using AppLauncher.ViewModels;

namespace AppLauncher.Services;

public class TrayIconService : IDisposable
{
    private TaskbarIcon? _trayIcon;
    private LauncherWindow? _window;

    public void Attach(LauncherWindow window)
    {
        _window    = window;
        _trayIcon  = (TaskbarIcon)Application.Current.Resources["TrayIcon"];
        _trayIcon.TrayMouseDoubleClick += OnTrayDoubleClick;
    }

    private void OnTrayDoubleClick(object sender, RoutedEventArgs e)
    {
        var vm = App.LauncherViewModel;
        if (vm == null) return;

        // 収納中は先に展開
        if (vm.IsStored)
        {
            App.SnapService?.ForceExpand(() => PinToggle(vm));
            return;
        }
        PinToggle(vm);
    }

    private static void PinToggle(LauncherViewModel vm)
    {
        _window?.Activate();
        vm.IsPinned = !vm.IsPinned;
    }

    public void Dispose() => _trayIcon?.Dispose();
}
```

```csharp
private void OnSnapRight (object s, RoutedEventArgs e) => ChangeSnap("right");

        ApplyNewSnap(direction);

    App.ConfigService.Current.Global.SnapPosition = direction;
```

### 12. StartupRegistryHelper（新規、小ヘルパー）

`Helpers/StartupRegistryHelper.cs` を新規作成する。

```csharp
using Microsoft.Win32;
using System.Reflection;

namespace AppLauncher.Helpers;

public static class StartupRegistryHelper
{
    private const string RunKey = @"SOFTWARE\Microsoft\Windows\CurrentVersion\Run";
    private const string AppName = "AppLauncher";

    public static void Apply(bool enable)
    {
        using var key = Registry.CurrentUser.OpenSubKey(RunKey, true);
        if (key == null) return;

        if (enable)
        {
            string? exePath = Assembly.GetExecutingAssembly().Location
                .Replace(".dll", ".exe");   // 単一 exe 発行時は .exe が正パス
            if (exePath != null)
                key.SetValue(AppName, $"\"{exePath}\"");
        }
        else
        {
            key.DeleteValue(AppName, throwOnMissingValue: false);
        }
    }
}
```
